YJ_count_all_factory = {
	every_country = {
		set_temp_variable = {
			var = YJ_civilian_factories
			value = num_of_controlled_factories
		}
		subtract_from_temp_variable = {
			var = YJ_civilian_factories
			value = num_of_military_factories
		}
		subtract_from_temp_variable = {
			var = YJ_civilian_factories
			value = num_of_naval_factories
		}
		add_to_temp_variable = {
			var = YJ_civilian_factories
			value = modifier@industrial_factory_donations
		}
		if = {
			limit = {
				has_country_flag = YJ_axis
			}
			add_to_variable = {
				var = global.YJ_axis_all_factory
				value = YJ_civilian_factories
			}
			add_to_variable = {
				var = global.YJ_axis_all_factory
				value = num_of_military_factories
			}
			add_to_variable = {
				var = global.YJ_axis_all_factory
				value = num_of_naval_factories
			}
			add_to_variable = {
				var = global.YJ_axis_arms_factory
				value = num_of_military_factories
			}
			add_to_variable = {
				var = global.YJ_axis_industrial_complex
				value = YJ_civilian_factories
			}
			add_to_variable = {
				var = global.YJ_axis_dockyard
				value = num_of_naval_factories
			}
		}

		if = {
			limit = {
				has_country_flag = YJ_allie
			}
			add_to_variable = {
				var = global.YJ_allie_all_factory
				value = YJ_civilian_factories
			}
			add_to_variable = {
				var = global.YJ_allie_all_factory
				value = num_of_military_factories
			}
			add_to_variable = {
				var = global.YJ_allie_all_factory
				value = num_of_naval_factories
			}
			add_to_variable = {
				var = global.YJ_allie_arms_factory
				value = num_of_military_factories
			}
			add_to_variable = {
				var = global.YJ_allie_industrial_complex
				value = YJ_civilian_factories
			}
			add_to_variable = {
				var = global.YJ_allie_dockyard
				value = num_of_naval_factories
			}
		}
	}
}

YJ_calculate_factory_balance_effect = {
	if = {
        limit = {
            date < 1938.6.1
        }

        set_variable = { global.YJ_axis_advantage = 0 }
        set_variable = { global.YJ_allie_advantage = 0 }
        set_variable = { global.YJ_axis_arms_advantage = 0 }
        set_variable = { global.YJ_allie_arms_advantage = 0 }
        set_variable = { global.YJ_axis_dockyard_advantage = 0 }
        set_variable = { global.YJ_allie_dockyard_advantage = 0 }
        set_variable = { global.YJ_axis_refit_cost = 0 }
        set_variable = { global.YJ_allie_refit_cost = 0 }
    }

	else = {
		set_variable = { global.YJ_axis_arms_advantage = global.YJ_allie_arms_factory }
		subtract_from_variable = { global.YJ_axis_arms_advantage = global.YJ_axis_arms_factory }
		clamp_variable = { var = global.YJ_axis_arms_advantage min = 0 }
		divide_variable = { global.YJ_axis_arms_advantage = 10000 }
		set_variable = { global.YJ_allie_arms_advantage = global.YJ_axis_arms_factory }
		subtract_from_variable = { global.YJ_allie_arms_advantage = global.YJ_allie_arms_factory }
		clamp_variable = { var = global.YJ_allie_arms_advantage min = 0 }
		divide_variable = { global.YJ_allie_arms_advantage = 10000 }

		set_variable = { global.YJ_axis_dockyard_advantage = global.YJ_allie_dockyard }
		subtract_from_variable = { global.YJ_axis_dockyard_advantage = global.YJ_axis_dockyard }
		clamp_variable = { var = global.YJ_axis_dockyard_advantage min = 0 }
		divide_variable = { global.YJ_axis_dockyard_advantage = 10000 }
		set_variable = { global.YJ_allie_dockyard_advantage = global.YJ_axis_dockyard }
		subtract_from_variable = { global.YJ_allie_dockyard_advantage = global.YJ_allie_dockyard }
		clamp_variable = { var = global.YJ_allie_dockyard_advantage min = 0 }
		divide_variable = { global.YJ_allie_dockyard_advantage = 10000 }

		set_variable = { global.YJ_axis_refit_cost = global.YJ_axis_dockyard_advantage }
		multiply_variable = { global.YJ_axis_refit_cost = -1 }
		set_variable = { global.YJ_allie_refit_cost = global.YJ_allie_dockyard_advantage }
		multiply_variable = { global.YJ_allie_refit_cost = -1 }

    	set_variable = { global.YJ_axis_advantage = global.YJ_allie_all_factory }
    	subtract_from_variable = { var = global.YJ_axis_advantage value = global.YJ_axis_all_factory }

    	if = {
       	 limit = { check_variable = { global.YJ_axis_advantage < 0 } }
       	 set_variable = { global.YJ_axis_advantage = 0 }
    	}

		divide_variable = { var = global.YJ_axis_advantage value = 500 }

			if = {
			limit = {
				check_variable = {
					global.YJ_axis_advantage > 0.5
				}
			}
			set_variable = {
				global.YJ_axis_advantage = 0.5
			}
		}

		set_variable = { global.YJ_allie_advantage = global.YJ_axis_all_factory }
		subtract_from_variable = { var = global.YJ_allie_advantage value = global.YJ_allie_all_factory }

		if = {
			limit = { check_variable = { global.YJ_allie_advantage < 0 } }
			set_variable = { global.YJ_allie_advantage = 0 }
		}

		divide_variable = { var = global.YJ_allie_advantage value = 500 }

			if = {
			limit = {
				check_variable = {
					global.YJ_allie_advantage > 0.5
				}
			}
			set_variable = {
				global.YJ_allie_advantage = 0.5
			}
		}
	}
}
